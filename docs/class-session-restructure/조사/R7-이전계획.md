핵심 결론 6줄.
1. **롤백 트랩은 DB 를 되돌리지 못한다.** 따라서 모든 스키마 변경은 "이전 릴리스 코드가 새 스키마에서 그대로 동작"해야 한다(추가형). 그 이상의 되돌림은 RDS 스냅샷/PITR 이다.
2. 구 앱은 새 서버에 붙으므로 **서버가 구 계약(`/sessions/*`, `host_id`, `access_code`, 세션 id 딥링크)을 계속 지켜야** 한다. 세션 id 가 그대로이므로 알림·초대·입장 자격·S3 키는 변환 없이 산다.
3. 분류는 **하나의 함수**(§4.1)로 만들어 백필 스크립트와 구 앱 호환 shim(`POST /sessions`에 `class_id` 없을 때)이 같은 규칙을 쓴다. 규칙이 갈리면 이전 직후 데이터와 이후 데이터가 다르게 묶인다.
4. 상담 묶기(같은 상담사·내담자의 과거 세션을 한 관계로)는 **회원 1명짜리만** 묶고 비회원은 1:1 로 둔다(비회원 동일인 판정 근거 없음 — 정책상 이름 일치로 인계 금지). 묶기 여부는 결정 후보 C-6.
5. 상담 이전은 **`client_counselor_links` 를 만들지 않는다**(채팅방·AI 비서 권한 부작용, R1 §9). 링크가 이미 있으면 연결만 하고 `ended` 면 관계를 `closed` 로 시작한다.
6. 백필은 SQL 한 트랜잭션 + `session_migration_map` 이 멱등 키 겸 역방향 근거다. dev 에서 `BEGIN … ROLLBACK` 으로 먼저 돌려 §8 검증 쿼리를 통과시킨 뒤에만 apply 한다.

### 3.2 type 분포를 알기 위한 조회 SQL 초안 (읽기 전용, 미실행)

원칙: 집계만(개수·분포), 이름·이메일·코드 값은 SELECT 하지 않는다 — `baseline-snapshot.sql` 선례. **ORM 의 숨김 필터를 우회하는 원시 SQL**이라 `hidden_at` 행도 센다.

```sql
-- D-1 유형 × 참여 방식 × 템플릿 × 숨김 × 상태
SELECT type, participant_mode, is_template, (hidden_at IS NOT NULL) AS hidden, status, count(*) AS sessions
FROM sessions GROUP BY 1,2,3,4,5 ORDER BY 1,2,3,4,5;

-- D-2 유형 × 활성 참여자 수 구간(호스트·제거·대기·대기취소 제외) × 비회원 포함
WITH a AS (
  SELECT s.id, s.type, s.participant_mode,
         count(p.id) FILTER (WHERE p.removed_at IS NULL AND p.waitlist_cancelled_at IS NULL
                               AND p.is_waitlisted = false AND p.user_id IS DISTINCT FROM s.host_id) AS n_active,
         count(p.id) FILTER (WHERE p.removed_at IS NULL AND p.waitlist_cancelled_at IS NULL
                               AND p.is_waitlisted = false AND p.user_id IS NULL) AS n_guest
  FROM sessions s LEFT JOIN session_participants p ON p.session_id = s.id
  WHERE s.is_template = false GROUP BY s.id, s.type, s.participant_mode)
SELECT type, participant_mode, least(n_active,3) AS active_bucket, (n_guest>0) AS has_guest, count(*) AS sessions
FROM a GROUP BY 1,2,3,4 ORDER BY 1,2,3,4;

-- D-3 custom 유형명 분포(상위 20, 자유 입력 라벨 — 집계 보고 전 개인명 포함 여부 확인)
SELECT custom_type_name, participant_mode, count(*) FROM sessions
WHERE type='custom' AND is_template=false GROUP BY 1,2 ORDER BY 3 DESC LIMIT 20;

-- D-4 상담 유형(clinical/hypnosis): (호스트, 회원 내담자) 쌍별 세션 수 히스토그램  → 묶기 효과 (C-6)
WITH c AS (
  SELECT s.host_id, p.user_id, count(*) AS n
  FROM sessions s JOIN session_participants p ON p.session_id = s.id
  WHERE s.type IN ('clinical','hypnosis') AND s.is_template=false AND p.user_id IS NOT NULL
    AND p.user_id <> s.host_id AND p.removed_at IS NULL AND p.waitlist_cancelled_at IS NULL AND p.is_waitlisted=false
  GROUP BY 1,2)
SELECT n AS sessions_per_pair, count(*) AS pairs FROM c GROUP BY 1 ORDER BY 1;

-- D-5 위 쌍 중 client_counselor_links 보유/상태
WITH c AS (SELECT DISTINCT s.host_id, p.user_id FROM sessions s JOIN session_participants p ON p.session_id=s.id
           WHERE s.type IN ('clinical','hypnosis') AND s.is_template=false AND p.user_id IS NOT NULL AND p.user_id<>s.host_id)
SELECT coalesce(l.status,'(링크 없음)') AS link_status, count(*) FROM c
LEFT JOIN client_counselor_links l ON l.client_id=c.user_id AND l.counselor_id=c.host_id GROUP BY 1;

-- D-6 같은 쌍의 세션이 여러 기관(organization_id)에 걸치는지(묶기 시 기관 혼재)
SELECT count(*) AS mixed_org_pairs FROM (
  SELECT s.host_id, p.user_id FROM sessions s JOIN session_participants p ON p.session_id=s.id
  WHERE s.type IN ('clinical','hypnosis') AND s.is_template=false AND p.user_id IS NOT NULL AND p.user_id<>s.host_id
  GROUP BY 1,2 HAVING count(DISTINCT s.organization_id) > 1) x;

-- D-7 호스트 계정 상태 × 활성 세션(미래 세션의 담당자가 비활성인지)
SELECT u.status AS host_status, s.status AS session_status, count(*) FROM sessions s JOIN users u ON u.id=s.host_id
WHERE s.is_template=false GROUP BY 1,2 ORDER BY 1,2;

-- D-8 코드 현황(값 비조회): 길이 × 상태
SELECT length(access_code) AS len, status, (hidden_at IS NOT NULL) AS hidden, count(*) FROM sessions
WHERE access_code IS NOT NULL AND is_template=false GROUP BY 1,2,3 ORDER BY 1,2,3;

-- D-9 딥링크가 세션 id 를 가리키는 영속 행 수
SELECT (SELECT count(*) FROM notifications WHERE extra->>'target_type' = 'session')   AS notif_session_links,
       (SELECT count(*) FROM agent_messages WHERE ref_type = 'session')                AS agent_session_refs,
       (SELECT count(*) FROM agent_delivery_logs)                                      AS agent_delivery_logs,
       (SELECT count(*) FROM session_email_invites)                                    AS email_invites;

-- D-10 이전 시점 점검: 진행 중/열린 세션(백필 락 시간·사용자 영향)
SELECT status, count(*) FROM sessions WHERE status IN ('open','in_progress') AND is_template=false GROUP BY 1;

-- D-11 정렬 키 결손(scheduled_at NULL=즉석 클래스) 개수 → seq 순서가 created_at 폴백인 비율
SELECT count(*) FILTER (WHERE scheduled_at IS NULL) AS no_schedule, count(*) AS total FROM sessions WHERE is_template=false;

-- D-12 템플릿 수와 보유자
SELECT count(*) AS templates, count(DISTINCT host_id) AS template_hosts FROM sessions WHERE is_template=true;
```

산출 후 해야 할 일: D-1·D-2 결과로 **결정 후보 C-4(다자 상담)·C-5(custom)·C-7(내담자 미상)** 의 실제 건수를 채우고, D-4~D-6 으로 C-6(묶기)의 효과를 Brian 에게 보여준다. → 이 숫자 없이 규칙을 확정하지 않는다.

### 4.1 분류 함수 (단일 규칙 — 백필 스크립트와 `POST /sessions` shim 이 공유)

입력: 세션 1건 + 활성 참여자 집합 `A` (호스트 본인·제거·대기·대기취소 제외, R1 §3.2 D-2 와 동일 정의).

| 순서 | 조건 | 상위 종류 | 규칙 코드 | 비고 |
|---|---|---|---|---|
| 0 | `is_template = true` | **이전 안 함** | — | §4.8 |
| 1 | `type = 'meditation'` | 명상 클래스 | **C1** | participant_mode 무관 |
| 2 | `type IN ('clinical','hypnosis')` 이고 `|A| ≥ 2` | 명상 클래스 | **C3** (유형 불일치 표시) | **결정 후보 C-4** |
| 3 | `type IN ('clinical','hypnosis')` 이고 `|A| = 1`, 회원 | 상담 관계(회원) | **R1** | 묶기 대상(§4.3) |
| 4 | 〃 `|A| = 1`, 비회원 | 상담 관계(비회원) | **R2** | 세션당 1관계 |
| 5 | 〃 `|A| = 0` | 상담 관계(내담자 미상, placeholder) | **R3** | **결정 후보 C-7** |
| 6 | `type = 'custom'` 이고 (`participant_mode='group'` 또는 `|A| ≥ 2`) | 명상 클래스 | **C2** | §4.4 |
| 7 | `type = 'custom'` 그 외 | 상담 관계 | R1/R2/R3 (3~5 와 동일) | §4.4 |

- **정원(`max_participants`)으로 그룹/1:1 을 판정하지 않는다.** 생성 스키마의 기본 정원이 10 이고 비명상 유형도 `max(입력,1)` 로 저장되므로(`schemas/session.py:28`, `services/session_service.py:362-366`) 상담 세션도 정원 10 일 수 있다.
- `participant_mode` 도 type 과 독립(`schemas/session.py:16,33`)이므로 type 규칙이 우선, custom 에서만 보조 지표로 쓴다.
- 이 함수는 **순수 함수**(DB 변경 없음)로 구현해 단위 테스트(표 8행 + 경계)로 고정한다.
- **R10 과의 차이(조정 필요)**: `R10-상담분리.md` §4 는 상담 유형이라도 `participant_mode='group'` 이면 집단 상담으로 보아 클래스로 보낸다. 이 문서는 Brian 규칙("clinical·hypnosis→상담")을 우선해 **참여자 수만** 본다. (clinical/hypnosis, `group`, 활성 참여자 ≤1) 건수를 D-1·D-2 로 확인해 갈리면 C-4 에서 함께 결정한다. 또 R10 은 상위를 `client_counselor_links` 확장(`sessions.counseling_link_id`)으로 보므로 이 문서의 `counseling_relationships`·`counseling_relationship_id` 는 그에 대응하는 이름이다(R1 §9.3). 어느 쪽이든 §4~§8 규칙은 이름 치환만으로 유효하되, **링크 확장을 택하면 이전분 링크 행이 채팅·AI 비서 접근을 부여하므로 `pending` 등 별도 상태로 시작해야 한다.**

### 4.2 명상 클래스 (C1·C2·C3): 세션 1건 → 클래스 1 + 세션 1

### 4.3 상담 (R1·R2·R3): 묶기 안 vs 1:1 안 비교

| 안 | 내용 | 장점 | 단점 / 위험 |
|---|---|---|---|
| **G-1. 1:1 매핑** | 상담 세션 1건 → 관계 1 + 세션 1 | 판단이 필요 없음, 결정론적, 역방향 쉬움 | "채용욱 상담의 4번째 세션" 이 되어야 할 기록이 관계 N개로 쪼개져 **새 IA(상담 관계 상세·추이)가 이전 데이터에서 무의미**. 상담사가 수동으로 병합할 도구가 필요 |
| **G-2. 쌍 묶기 (기본값)** | 활성 참여자가 **회원 정확히 1명**인 상담 세션을 `(host_id, 회원 user_id)` 쌍으로 한 관계에 묶음. `seq` = `coalesce(scheduled_at, started_at, created_at)`, `created_at`, `id` 순 | 새 IA 의 핵심 가치(관계 단위 이력)가 이전 직후부터 의미 있음. 회원은 `user_id` 로 동일인 판정 가능 | ① 같은 쌍이 **여러 기관**에 걸치면 혼재(D-6 로 건수 확인; 기본: 최근 세션의 기관 사용 + 보고) ② 상담사가 의도적으로 분리한 건(다른 사유의 상담)을 묶음 ③ 링크 없는 쌍도 관계가 생김 |
| G-3. 상담사 확인형 | 후보 묶음을 제안하고 상담사가 승인 | 가장 정확 | 이전 시점에 사람 개입 필요, 지연되면 이전 불가 |

**기본값: G-2(회원만 묶기) + 비회원은 G-1.** 이유: 비회원은 세션 간 동일인 증거가 없고 정책상 이름 일치만으로 인계하지 않는다(`policy-revision-r4.md` 위임받은 기술 결정). → **결정 후보 C-6**(묶기 여부). G-3 은 후속 도구로 "관계 병합/분리" 기능을 제공해 보완.

**내담자 여럿/비회원/0명 처리**

| 케이스 | 처리 | 규칙 |
|---|---|---|
| 활성 참여자 정확히 1명, 회원 | `(host, member)` 관계에 묶음. 이미 활성 관계가 있으면 그 관계에 추가(멱등) | R1 |
| 활성 참여자 정확히 1명, 비회원 | 세션당 `counseling_guest_clients` 1행(참여자 행의 이름·성별·생년월일·리포트 이메일 **복사**, `email_verified_at=NULL`, `origin='migrated'`, 스크럽된 행은 `pii_scrubbed_at` 도 복사) + 관계 1행. 참여자 행에 `guest_client_id` 연결 | R2 |
| 활성 참여자 2명 이상 (상담 type) | **명상 클래스로 이전**, `map.flags.legacy_type_mismatch=true`, `sessions.type` 은 그대로 보존 → **결정 후보 C-4** | C3 |
| 활성 참여자 0명 | 상담사당 **placeholder 관계 1개**(`is_placeholder=true`, 내담자 NULL)에 모음. 삭제하지 않음 | R3 → **결정 후보 C-7** |
| 호스트가 참여자 행에 같이 있음(자기 테스트) | 호스트 행은 `A` 에서 제외 | |
| 제거된 참여자(`removed_at`)만 있는 상담 세션 | `|A|=0` 이라 R3 로 가지만 제거된 참여자 행·리포트는 보존됨. 건수는 D-2 와 별도 확인 필요 | `미확인` |
| 링크(`client_counselor_links`) | **생성하지 않는다.** 있으면 `link_id` 연결, `status='ended'` 이고 활성 세션이 없으면 관계 `closed`(`closed_at=link.ended_at`), 활성 세션이 있으면 `active` 유지 | R1 §9 |

### 4.4 `custom` 유형 이전 기본값

- 규칙 6·7(§4.1): **그룹(`participant_mode='group'`) 또는 활성 참여자 2명 이상이면 명상 클래스, 아니면 상담** 으로 보낸다. 정원 값은 쓰지 않는다.
- `custom_type_name`: 클래스로 가면 `classes` 의 분류 라벨 후보(컬럼 신설은 구현 단계에서 결정, 지금은 세션에 유지), 상담으로 가면 세션에 유지. 분류 결과와 라벨을 `session_migration_map.flags` 에 기록해 사후 재분류 근거로 남긴다.
- 기본값이 틀릴 수 있으므로 D-3 로 실제 라벨 분포를 보고 Brian 이 **라벨별 지정표**(예: "마음챙김 산책" → 명상)를 줄 수 있게 한다 — **결정 후보 C-5**. 지정표는 `custom_type_name` 일치 시 규칙보다 우선(`rule='C2-manual'` 또는 `'R*-manual'`).
- 이후 신규 생성에서 `custom` 을 계속 허용할지(기획 단계)는 별개 결정.
- R10 §4 는 `custom_type_name` **키워드 규칙**(명상·마음챙김·호흡·요가·MBSR → 명상 / 상담·치료·코칭·심리·면담 → 상담, 충돌은 "분류 확인 필요" 목록)을 제안한다. 이 문서의 라벨별 지정표의 **자동 초안**으로 쓰기에 적합하다(키워드가 충돌하거나 규칙 결과와 방식이 어긋나면 사람 확인). D-3 의 실제 라벨 분포를 본 뒤 확정.

### 4.5 `host_id` → 강사/진행 강사 자동 매핑

| 대상 | 규칙 | 근거 |
|---|---|---|
| `sessions.host_id` | **변경 없음** — "진행 강사/진행 상담사"로 의미만 확정 | R1 §13 H1 |
| 명상: `class_instructors` | 클래스마다 `(class_id, host_id, role='lead', status='active', added_by=host_id, added_at=세션 created_at)` 1행. 복합 FK 의 참조 대상이므로 **세션 갱신(`class_id` 채움) 전에 먼저 INSERT** | R1 §12 |
| 상담: `counseling_relationship_counselors` | 관계마다 `(relationship_id, counselor_id=host_id, 'active', from_at=started_at)` 1행 | R1 §8.5 |
| 개설자 | `classes.created_by` ← `host_id`, 상담 `counselor_id` ← `host_id` | |
| 비활성 호스트(`users.status ≠ active`) | **그대로 lead 로 등록**(이력 충실). 미래 세션(`ready|scheduled|open`)이면 `needs_reassign` 후보로 **보고만**(D-7), 상태 변경은 R9 규칙 도입 후 | R9 §5 |
| 기관 | `organization_id`·`organization_attribution_known` 값 복사 — **추정으로 채우지 않는다** | `models/session.py:50-51` 주석 |

### 4.6 참여자 → 멤버십/출석

- 명상 클래스로 간 세션: 참여자 행마다 `class_members` 1행을 만들고 상태를 **R4 §10 표**대로 백필(`removed_at`→removed/withdrawn, `waitlist_cancelled_at`→left/withdrawn, `is_waitlisted`→waitlisted/expected, `entered_at`→attended 등). `entered_at` 도입 전 데이터는 입장 여부를 모르므로 결석으로 단정하지 않고 `expected` 로 둔다(R4 §10).
- `session_participants` **행은 변경·삭제하지 않는다**(id 불변). `member_id`(·`guest_client_id`) 컬럼만 채운다.
- 상담 세션: 멤버십 테이블 없음. 회원은 `client_user_id`, 비회원은 `guest_client_id`.

### 4.8 설정 복사·템플릿

- 설정 컬럼은 **복사만**(상위 기본값 + 세션 유효값, P1). 불일치 시 세션 값이 진실.
- 템플릿(`is_template=true`): **이전 대상 아님.** 부모 FK 를 비워 둔 채 `sessions` 에 남기고 CHECK 예외(`is_template OR …`)로 통과. M6 에서 `class_templates` 로 복사 후 정리. 상담 템플릿 필요 여부 `미확인`.
- 모든 이전 행은 `session_migration_map` 에 `(session_id PK, parent_kind, parent_id, rule, flags)` 기록 — 멱등 키, 감사, 역방향 근거.

- 상담 초대(관계 단위 서명 링크)는 신규 경로이며 구 앱과 무관.

### 6.3 알림 딥링크
- 인앱 알림 `extra`: `target_type='session'`, `target_id=<세션 id>`, 레거시 루트 키 `session_id` (`services/notification_service.py:86-106`, 발화 `session_service.py:1181-1194`). 클릭 이동은 프론트가 `target_id` 로 `/sessions/:id`·`/app/sessions/:id`·`/class/:id/enter` 를 만든다(`frontend/src/lib/api/notifications.ts:157-170`).
- AI 비서 CTA·브리핑에 박힌 URL: `/app/sessions/{id}`(`services/agent_service.py:504`), `/sessions/{id}`(`services/agent_briefing.py:360`)는 **DB 에 영구 저장**된다(`agent_messages.cta`).
- 전략: 세션 id 불변이므로 데이터 변환 불필요. **라우트 `/sessions/:id`·`/app/sessions/:id`·`/class/:id/enter` 는 폐기하지 말고** 새 라우트로 리다이렉트(예: `/classes/:class_id?session=:id`)로 유지. 신규 알림은 `params` 에 클래스 정보를 가산하되 `target_id`(세션 id)와 `session_id` 레거시 키는 계속 채운다(R2 §11-4).
- 푸시: 세션 이벤트 전용 푸시 딥링크는 발견되지 않음(채팅·AI 비서 딥링크만, `chat_service.py:1432`, `agent_service.py:304`, `tasks/push_task.py:165`) — 세션 푸시 경로 `미확인`.
- 영속 행 건수는 D-9 로 확인(이전 전후 동일해야 함, V17).

### 6.5 구 앱(Capacitor)·구 웹 탭
- 구 클라이언트가 호출하는 `/sessions/*` 61개(R2 §3.1) **전부 유지**, 응답에 `class_id`·`class_title`·`seq` 등 **추가 필드만**. 필드 삭제·의미 변경 금지(`SessionResponse.host_id`, `max_participants`, `access_code` 등 계속 채움).
- 구 앱이 클래스 단위 필드를 `PUT /sessions/{id}` 로 수정하면: 해당 클래스가 `origin ∈ {migrated, shim}` 이고 세션이 1개뿐일 때만 **클래스 기본값에도 동기화**(write-through). 세션이 2개 이상인 클래스(새 앱이 만든 것)는 세션 값만 수정.
- 구 앱이 `DELETE /sessions/{id}`(숨김)한 세션이 클래스의 **마지막 가시 세션**이면 클래스도 `hidden_at`(빈 클래스가 새 앱에 남지 않게).
- 구 앱의 `POST /sessions/{id}/new-run`·`/duplicate`·`/save-as-template`·`GET /sessions/templates`: 유지. `duplicate` 산출물은 shim 이 새 클래스(`origin='shim'`)로 만든다.
- 앱 종료 판단 지표: shim 경로 호출(클래스 없는 `POST /sessions`)·레거시 라우트 호출 건수를 구조화 로그로 남겨 M5 종료 판단 근거로 쓴다(기술 제안).

### 6.6 읽는 쪽 전환 순서
1. 새 코드는 **부모를 쓰되 세션 컬럼도 계속 채운다**(dual-write, P1) — 엔진은 `sessions` 를 계속 읽음.
2. 신규 화면만 부모를 읽는다. 목록 API 는 구 응답 형태를 유지하면서 부모 필드를 가산.
3. 세션 컬럼 쓰기 중단은 M6 이후 별도 결정.

### 6.7 상담 영역 호환
- 이전된 상담 세션은 구 앱에서 **그대로 세션으로 보인다**(관계 개념은 새 앱만). 상담 관계 상세는 새 앱만.
- 이전된 비회원 상담 관계는 `reentry_secret_hash` 가 없으므로 **재방문 불가**(초대 재발송 필요) — 이는 정상이며 UI 안내 문구 대상 → **결정 후보 C-14**.

### 6.8 `POST /sessions` shim (구 앱 신규 생성)
- `class_id`·`counseling_relationship_id` 없는 생성 요청은 서버가 §4.1 함수로 분류해 **부모를 자동 생성**한다: 명상→클래스(`origin='shim'`, `class_instructors` lead), 상담→`(host, 참여자)` 쌍의 활성 관계 재사용 또는 신규(참여자 없으면 placeholder).
- 이 함수가 백필과 같으므로 M2~M4 사이에 생긴 스트레이도 같은 결과로 정리된다.

### 7.1 절차

4. **스크립트화**: `backend/scripts/` 에 `reissue_class_codes.py` 와 같은 형태(기본 dry-run, `--apply`, JSON 건수 출력, 멱등)로 래핑. SQL 은 서비스 코드에 넣지 말고 분류 함수만 Python 으로 공유(§4.1).
5. **apply**: Brian 승인 후 `LOCK TABLE … SHARE ROW EXCLUSIVE` 안에서 §7.2 실행 → §8 재실행 → 스냅샷 ID·결과를 `summary` 에 기록.
6. **재실행**: M4 직후·M6 직전에 같은 스크립트 재실행(스트레이 흡수). 결과 건수가 0 이어야 한다.

### 7.2 백필 SQL 초안 (PostgreSQL, **미실행**)

```sql
BEGIN;
LOCK TABLE sessions IN SHARE ROW EXCLUSIVE MODE;            -- 읽기 허용, 쓰기 수 초 차단
LOCK TABLE session_participants IN SHARE ROW EXCLUSIVE MODE;

-- S0. 분류 임시표 (멱등: 부모도 매핑도 없는 비템플릿 세션만)
CREATE TEMP TABLE _cls ON COMMIT DROP AS
SELECT s.id AS session_id, s.host_id, s.type, s.participant_mode, s.status, s.created_at,
       s.organization_id, s.organization_attribution_known,
       coalesce(s.scheduled_at, s.started_at, s.created_at) AS ord_at,
       a.n_active, a.n_member, a.n_guest, a.member_id,
       CASE
         WHEN s.type = 'meditation'                                            THEN 'class'
         WHEN s.type IN ('clinical','hypnosis') AND a.n_active >= 2            THEN 'class'   -- C-4
         WHEN s.type = 'custom' AND (s.participant_mode = 'group' OR a.n_active >= 2) THEN 'class'  -- C-5
         ELSE 'counseling'
       END AS parent_kind
FROM sessions s
JOIN LATERAL (
  SELECT count(*) AS n_active,
         count(*) FILTER (WHERE p.user_id IS NOT NULL) AS n_member,
         count(*) FILTER (WHERE p.user_id IS NULL)     AS n_guest,
         (array_agg(p.user_id) FILTER (WHERE p.user_id IS NOT NULL))[1] AS member_id
  FROM session_participants p
  WHERE p.session_id = s.id AND p.removed_at IS NULL AND p.waitlist_cancelled_at IS NULL
    AND p.is_waitlisted = false AND p.user_id IS DISTINCT FROM s.host_id
) a ON true
WHERE s.is_template = false AND s.class_id IS NULL AND s.counseling_relationship_id IS NULL
  AND NOT EXISTS (SELECT 1 FROM session_migration_map m WHERE m.session_id = s.id);

-- S1-a. 명상: 세션당 클래스 1개 배정 (매핑이 멱등 키)
INSERT INTO session_migration_map (session_id, parent_kind, parent_id, rule, flags)
SELECT session_id, 'class', gen_random_uuid(),
       CASE WHEN type='meditation' THEN 'C1' WHEN type='custom' THEN 'C2' ELSE 'C3' END,

-- S3. 세션에 부모·seq 연결 (재실행 시 기존 max(seq) 뒤에 이어 붙임)
WITH mm AS (
  SELECT m.session_id, m.parent_kind, m.parent_id,
         coalesce(s.scheduled_at, s.started_at, s.created_at) AS ord_at, s.created_at,
         (SELECT coalesce(max(q.seq),0) FROM sessions q
           WHERE q.class_id = m.parent_id OR q.counseling_relationship_id = m.parent_id) AS base
  FROM session_migration_map m JOIN sessions s ON s.id = m.session_id
  WHERE s.is_template=false AND s.class_id IS NULL AND s.counseling_relationship_id IS NULL),
n AS (SELECT *, base + row_number() OVER (PARTITION BY parent_id ORDER BY ord_at, created_at, session_id) AS seq FROM mm)
UPDATE sessions s
SET class_id = CASE WHEN n.parent_kind='class' THEN n.parent_id END,
    counseling_relationship_id = CASE WHEN n.parent_kind='counseling' THEN n.parent_id END,
    seq = n.seq
FROM n WHERE s.id = n.session_id;

-- S4. 명상 멤버십 (R4 §10 상태 백필). 참여자 id 는 그대로, member_id 만 연결

            WHEN p.waitlist_cancelled_at IS NOT NULL THEN 'left'
            WHEN p.is_waitlisted THEN 'waitlisted'
            WHEN s.status = 'completed' THEN 'completed'
            ELSE 'active' END,
       'migration', p.guest_name, p.gender, p.birth_date, p.report_email, p.pii_scrubbed_at, p.joined_at, s.id, now(), now()

UPDATE session_participants p SET member_id = pm.member_id FROM _pm pm WHERE p.id = pm.participant_id;
-- attendance_status 백필(entered_at 있으면 attended, 제거/대기취소 withdrawn, 그 외 expected)은 R4 §10 규칙대로 별도 UPDATE
```

주의: PostgreSQL 에는 `max(uuid)` 집계가 없어 `::text` 경유로 썼다 — 구현 시 `(array_agg(...))[1]` 로 정리해도 된다. `lower(btrim())` 는 앱의 이메일 정규화 규칙과 같은지 `미확인`. 문법·실행은 검증하지 않은 초안이다(DB 접속 금지).

### 7.3 멱등성·동시성
- 멱등 키 = `session_migration_map.session_id`(PK). 재실행은 매핑이 없는 비템플릿 세션만 처리한다.
- 동시 쓰기: `LOCK TABLE sessions IN SHARE ROW EXCLUSIVE MODE` 로 세션 INSERT/UPDATE 를 수 초 막는다(읽기 허용). 규모가 작아(§3.1) 허용 가능.
- 재개한 상담 관계에 새 세션이 붙을 때 `base` 가 기존 `max(seq)` 라 번호가 이어진다.

### 7.4 데이터 역방향(보조 수단, 정상 롤백 아님)
정상 롤백은 **코드 revert 뿐**이고 DB 는 유지한다(§2). 데이터를 정말 되돌려야 하면 map 기반으로:

```sql
BEGIN;
CREATE TEMP TABLE _rb ON COMMIT DROP AS
SELECT m.session_id, m.parent_kind, m.parent_id FROM session_migration_map m
WHERE NOT EXISTS (   -- 이전 후 새 코드가 부모에 붙인 세션이 있으면 제외(데이터 보호)
  SELECT 1 FROM sessions s2 WHERE (s2.class_id=m.parent_id OR s2.counseling_relationship_id=m.parent_id)
    AND s2.id NOT IN (SELECT session_id FROM session_migration_map));
UPDATE session_participants SET member_id=NULL, guest_client_id=NULL WHERE session_id IN (SELECT session_id FROM _rb);
UPDATE sessions SET class_id=NULL, counseling_relationship_id=NULL, seq=NULL WHERE id IN (SELECT session_id FROM _rb);
DELETE FROM class_members WHERE class_id IN (SELECT parent_id FROM _rb WHERE parent_kind='class');
DELETE FROM class_instructors WHERE class_id IN (SELECT parent_id FROM _rb WHERE parent_kind='class');
DELETE FROM classes WHERE id IN (SELECT parent_id FROM _rb WHERE parent_kind='class') AND origin='migrated';
DELETE FROM counseling_relationship_counselors WHERE relationship_id IN (SELECT parent_id FROM _rb WHERE parent_kind='counseling');
DELETE FROM counseling_relationships r WHERE r.id IN (SELECT parent_id FROM _rb WHERE parent_kind='counseling')
  AND NOT EXISTS (SELECT 1 FROM sessions s WHERE s.counseling_relationship_id = r.id);
DELETE FROM counseling_guest_clients WHERE origin='migrated' AND id NOT IN (SELECT client_guest_id FROM counseling_relationships WHERE client_guest_id IS NOT NULL);
DELETE FROM session_migration_map WHERE session_id IN (SELECT session_id FROM _rb);
COMMIT;   -- 복합 FK 가 걸린 뒤에는 sessions 갱신을 먼저(위 순서)
```
- 이 SQL 은 **이전 직후(새 코드 배포 전)** 에만 안전하다. 새 코드 배포 이후엔 부모에 새 세션·멤버가 붙어 제외 규칙이 작동하며 사실상 불가 → 그때는 스냅샷/PITR.

## 8. 검증 쿼리 초안 (이전 전/후 비교, 모두 읽기 전용)

> "기대"는 이전 직후(M2) 기준. 새 코드가 세션을 추가한 뒤에는 V4 같은 1:1 가정 쿼리를 쓰지 않는다.

```sql
-- V1 type 별 건수 대조 — D-1 의 이전 전 집계와 합이 같아야 하고, 규칙 코드별로 기대와 일치
SELECT s.type, m.parent_kind, m.rule, count(*) FROM sessions s
LEFT JOIN session_migration_map m ON m.session_id = s.id
WHERE s.is_template=false GROUP BY 1,2,3 ORDER BY 1,2,3;
-- 기대: meditation→class/C1 전부 · clinical,hypnosis→counseling/R1|R2|R3 또는 class/C3 · custom→C2 또는 R*

-- V2 고아/스트레이 세션: 부모가 없거나 둘 다 있음  (기대 0)
SELECT count(*) FROM sessions s WHERE s.is_template=false AND (s.class_id IS NULL) = (s.counseling_relationship_id IS NULL);
SELECT count(*) FROM sessions s LEFT JOIN session_migration_map m ON m.session_id=s.id
 WHERE s.is_template=false AND m.session_id IS NULL;                       -- 매핑 없음(기대 0, M4 이후엔 shim 산출 별도)

-- V3 고아 부모 (기대 0; placeholder 제외)
SELECT count(*) FROM classes c WHERE NOT EXISTS (SELECT 1 FROM sessions s WHERE s.class_id=c.id);
SELECT count(*) FROM counseling_relationships r WHERE NOT r.is_placeholder
   AND NOT EXISTS (SELECT 1 FROM sessions s WHERE s.counseling_relationship_id=r.id);

-- V4 (M2 직후만) 이전 클래스는 세션 정확히 1개 (기대 0행)
SELECT c.id FROM classes c JOIN sessions s ON s.class_id=c.id WHERE c.origin='migrated' GROUP BY c.id HAVING count(*) <> 1;

-- V5 seq 중복·결번 (기대 0행)

 WHERE seq IS NOT NULL GROUP BY 1,2 HAVING count(*) > 1;
SELECT parent FROM (SELECT coalesce(class_id, counseling_relationship_id) AS parent, min(seq) mn, max(seq) mx, count(*) c
                    FROM sessions WHERE seq IS NOT NULL GROUP BY 1) x WHERE mn <> 1 OR mx <> c;

-- V6 참여자 수 대조
SELECT (SELECT count(*) FROM session_participants)                                         AS participants_total,   -- 이전 전과 동일해야 함
       (SELECT count(*) FROM session_participants p JOIN sessions s ON s.id=p.session_id WHERE s.class_id IS NOT NULL) AS in_class_sessions,
       (SELECT count(*) FROM class_members)                                                AS members,              -- = in_class_sessions (M2 직후)
       (SELECT count(*) FROM session_participants WHERE member_id IS NOT NULL)             AS linked;               -- = members
SELECT (user_id IS NULL) AS guest, is_waitlisted, (removed_at IS NOT NULL) AS removed,
       (waitlist_cancelled_at IS NOT NULL) AS wl_cancel, count(*) FROM session_participants GROUP BY 1,2,3,4 ORDER BY 1,2,3,4;
SELECT kind, status, count(*) FROM class_members GROUP BY 1,2 ORDER BY 1,2;   -- 위 분포와 R4 §10 표로 대응

-- V7 host → 강사/상담사 행 존재 (복합 FK VALIDATE 사전 검사, 기대 0)
SELECT count(*) FROM sessions s WHERE s.class_id IS NOT NULL AND NOT EXISTS
  (SELECT 1 FROM class_instructors i WHERE i.class_id=s.class_id AND i.user_id=s.host_id);
SELECT count(*) FROM sessions s WHERE s.counseling_relationship_id IS NOT NULL AND NOT EXISTS
  (SELECT 1 FROM counseling_relationship_counselors rc WHERE rc.relationship_id=s.counseling_relationship_id AND rc.counselor_id=s.host_id);

-- V8 코드 대조 (기대 0 / 0행)
SELECT count(*) FROM classes c JOIN sessions s ON s.class_id=c.id
 WHERE c.origin='migrated' AND (c.access_code IS DISTINCT FROM s.access_code OR c.legacy_access_code IS DISTINCT FROM s.legacy_access_code);
SELECT access_code, count(*) FROM classes WHERE access_code IS NOT NULL AND hidden_at IS NULL
   AND status NOT IN ('completed','cancelled','archived') GROUP BY 1 HAVING count(*) > 1;   -- 코드 값 노출 주의: 건수만 보고

-- V9 설정 복사 대조 (기대 0)
SELECT count(*) FROM classes c JOIN sessions s ON s.class_id=c.id WHERE c.origin='migrated' AND (
  c.capacity IS DISTINCT FROM greatest(s.max_participants,1) OR c.participant_mode IS DISTINCT FROM s.participant_mode
  OR c.linkband_mode IS DISTINCT FROM s.linkband_mode OR c.location_type IS DISTINCT FROM s.location_type
  OR c.default_duration_min IS DISTINCT FROM s.duration_min OR c.record_audio IS DISTINCT FROM s.record_audio
  OR c.record_video IS DISTINCT FROM s.record_video OR c.reminder_offsets IS DISTINCT FROM s.reminder_offsets);

-- V10 상태·숨김·기관 정합 (기대 0)
SELECT count(*) FROM classes c JOIN sessions s ON s.class_id=c.id WHERE c.origin='migrated' AND (
  (s.status='completed' AND c.status<>'completed') OR (s.status='cancelled' AND c.status<>'cancelled')
  OR ((s.hidden_at IS NULL) <> (c.hidden_at IS NULL))
  OR c.organization_id IS DISTINCT FROM s.organization_id OR c.organization_attribution_known <> s.organization_attribution_known);

-- V11 제약 도입 가능성 사전 검사 (M3, 기대 0)
SELECT count(*) FROM sessions WHERE class_id IS NOT NULL AND counseling_relationship_id IS NOT NULL;

-- V12 상담 규칙 위반: 관계의 내담자와 세션 활성 참여자가 다름 / placeholder 인데 참여자 있음 (기대 0)
SELECT count(*) FROM counseling_relationships r JOIN sessions s ON s.counseling_relationship_id=r.id
 JOIN session_participants p ON p.session_id=s.id AND p.removed_at IS NULL AND p.waitlist_cancelled_at IS NULL
      AND p.is_waitlisted=false AND p.user_id IS DISTINCT FROM s.host_id
 WHERE (NOT r.is_placeholder AND r.client_user_id IS NOT NULL AND p.user_id IS DISTINCT FROM r.client_user_id)
    OR (r.is_placeholder);

FROM sessions s LEFT JOIN session_migration_map m ON m.session_id=s.id WHERE s.is_template=false GROUP BY 1,2,3 ORDER BY 1,2,3;

SELECT 'research_consent_events', count(*) FROM research_consent_events UNION ALL SELECT 'research_export_batches', count(*) FROM research_export_batches UNION ALL
SELECT 'data_export_jobs', count(*) FROM data_export_jobs UNION ALL SELECT 'data_export_audits', count(*) FROM data_export_audits;

-- V17 영속 딥링크의 세션 참조가 살아 있는지 (이전 전/후 동일해야 함)
```
